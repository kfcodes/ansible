===== roles/system_dev/tasks/app_setup.yml =====
- name: Create application directories
  file:
    path: "/home/{{ local_usr }}/{{ item.path }}"
    mode: "{{ item.attrs }}"
    state: directory
    recurse: true
    owner: "{{ local_usr }}"
    group: "{{ local_usr }}"
  loop: "{{ application_directories }}"
  become_user: "{{ local_usr }}"

- name: Install NPM packages for reactapp (if path exists)
  community.general.npm:
    path: "/home/{{ local_usr }}/.reactapp/"
    state: latest
  when: ansible_facts['user_dir'] is defined and ansible_facts['user_dir'] is directory
  ignore_errors: true
  become_user: "{{ local_usr }}"

- name: Install Python requirements for fastAPI (if path exists)
  pip:
    requirements: /home/{{ local_usr }}/.fastapi/requirements.txt
    virtualenv: /home/{{ local_usr }}/.fastapi/.venv
  when: ansible_facts['user_dir'] is defined and ansible_facts['user_dir'] is directory
  ignore_errors: true
  become_user: "{{ local_usr }}"

- name: Generate global .gitconfig from template
  template:
    src: "gitconfig.j2"
    dest: "/home/{{ local_usr }}/.gitconfig"
    owner: "{{ local_usr }}"
  become_user: "{{ local_usr }}"

===== roles/system_dev/tasks/clone_repositories.yml =====
- name: Pull development repositories
  git:
    repo: "{{ item.repo }}"
    dest: "/home/{{ local_usr }}/{{ item.dest }}"
    version: "{{ item.version | default('HEAD') }}"
    accept_hostkey: true
  loop: "{{ dev_repositories }}"
  become_user: "{{ local_usr }}"

===== roles/system_dev/tasks/github_ssh.yml =====
---
- name: Ensure user .ssh directory exists
  ansible.builtin.file:
    path: "/home/{{ local_usr }}/.ssh"
    state: directory
    owner: "{{ local_usr }}"
    group: "{{ local_usr }}"
    mode: "0700"

- name: Check GitHub SSH private key exists
  ansible.builtin.stat:
    path: "/home/{{ local_usr }}/.ssh/{{ devtools_github_ssh_key }}"
  register: github_ssh_private_key

- name: Fail if GitHub SSH private key is missing
  ansible.builtin.fail:
    msg: "Missing GitHub SSH private key: /home/{{ local_usr }}/.ssh/{{ devtools_github_ssh_key }}"
  when: not github_ssh_private_key.stat.exists

- name: Set GitHub SSH private key permissions
  ansible.builtin.file:
    path: "/home/{{ local_usr }}/.ssh/{{ devtools_github_ssh_key }}"
    owner: "{{ local_usr }}"
    group: "{{ local_usr }}"
    mode: "0600"

- name: Check GitHub SSH public key exists
  ansible.builtin.stat:
    path: "/home/{{ local_usr }}/.ssh/{{ devtools_github_ssh_key }}.pub"
  register: github_ssh_public_key

- name: Set GitHub SSH public key permissions
  ansible.builtin.file:
    path: "/home/{{ local_usr }}/.ssh/{{ devtools_github_ssh_key }}.pub"
    owner: "{{ local_usr }}"
    group: "{{ local_usr }}"
    mode: "0644"
  when: github_ssh_public_key.stat.exists

- name: Ensure user SSH config exists
  ansible.builtin.file:
    path: "/home/{{ local_usr }}/.ssh/config"
    state: touch
    owner: "{{ local_usr }}"
    group: "{{ local_usr }}"
    mode: "0600"

- name: Configure github.com SSH host
  ansible.builtin.blockinfile:
    path: "/home/{{ local_usr }}/.ssh/config"
    create: true
    owner: "{{ local_usr }}"
    group: "{{ local_usr }}"
    mode: "0600"
    marker: "# {mark} ANSIBLE MANAGED BLOCK - github.com ssh"
    block: |
      Host {{ devtools_github_ssh_host }}
          HostName {{ devtools_github_ssh_hostname }}
          User {{ devtools_github_ssh_user }}
          IdentityFile ~/.ssh/{{ devtools_github_ssh_key }}
          IdentitiesOnly yes

- name: Ensure user gitconfig exists before repository cloning
  ansible.builtin.file:
    path: "/home/{{ local_usr }}/.gitconfig"
    state: touch
    owner: "{{ local_usr }}"
    group: "{{ local_usr }}"
    mode: "0644"

- name: Rewrite GitHub HTTPS URLs to SSH before repository cloning
  ansible.builtin.blockinfile:
    path: "/home/{{ local_usr }}/.gitconfig"
    create: true
    owner: "{{ local_usr }}"
    group: "{{ local_usr }}"
    mode: "0644"
    marker: "# {mark} ANSIBLE MANAGED BLOCK - github url rewrite"
    block: |
      [url "git@github.com:"]
          insteadOf = https://github.com/
  when: devtools_github_rewrite_https_to_ssh | bool
